# 通用任务停止 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增通用「停止」能力：看板批次级 + 运行行级停止在途任务，落 `skipped`（error「手动停止」）、不重试、不计熔断。

**Architecture:** 进程内取消标记 `CancelRegistry`（按 `profileId:taskKey`）+ `TaskStopper` 编排（置标记 → 结算非终态行为 skipped → 关窗中止运行中会话 → 清开窗登记）。`WindowRunner` 在「会话开始前」与「任务失败 catch」两处消费标记；`scheduleRetry`/重试恢复回调遇标记跳过入队。不改运行状态机、不新增状态、不做队列手术式移除。

**Tech Stack:** TypeScript（严格模式）、Express 5、Node SQLite（libsql）、React 18 + antd 5 + react-query、vitest + supertest + @testing-library/react。

## Global Constraints

- 代码风格：无分号、单引号、2 空格缩进、TS 严格模式；文件头中文注释块说明职责与依赖方向。
- 注释 / 文档 / commit message 一律中文；commit 风格 conventional：`feat:`/`docs:` + 中文描述。
- 依赖方向不可反向：`tasks → engine → {integrations, automation} → infrastructure`；`server → {engine, infrastructure}`；前端发请求只走 `web/src/api/endpoints.ts`。
- 停止后复用 `skipped` 终态，`error` 文案固定为 `手动停止`；不新增状态、不改 `VALID_TRANSITIONS`。
- 停止只作用于**非终态**运行行（`pending`/`running`/`retry_wait`）；终态行不动、返回未停止。
- 每批改完必须 `npm run typecheck`、`npm test` 通过；前端改动另跑 `npm run test:web`。
- 本会话不逐任务提交（一次提交在末尾，由用户决定）；实现只改代码。

创建 `src/engine/cancel-registry.ts`：

```ts
/**
 * 取消标记注册表（engine 层）：按 (窗口, 任务) 记录「已请求停止」，供 runner / 重试回调消费
 * 依赖方向：无依赖，被 window-runner / task-stopper / app 装配使用
 * 设计思路：进程内内存态（重启即清空）；request 置位，消费方 isCancelled 判定后 clear
 */
export class CancelRegistry {
  private readonly keys = new Set<string>()

  private static key(profileId: number, taskKey: string): string {
    return `${profileId}:${taskKey}`
  }

  /** 请求停止某窗口某任务（置位） */
  request(profileId: number, taskKey: string): void {
    this.keys.add(CancelRegistry.key(profileId, taskKey))
  }

  /** 该窗口该任务是否已请求停止 */
  isCancelled(profileId: number, taskKey: string): boolean {
    return this.keys.has(CancelRegistry.key(profileId, taskKey))
  }

  /** 消费 / 清理标记（会话跳过、任务中止结算、重试跳过时调用） */
  clear(profileId: number, taskKey: string): void {
    this.keys.delete(CancelRegistry.key(profileId, taskKey))
  }
}
```

创建 `src/engine/task-stopper.ts`：

```ts
/**
 * 任务停止器（engine 层）：停止某 (窗口, 任务) 当天的在途运行
 * 依赖方向：依赖 infrastructure/db 类型与 CancelRegistry，被 app 装配、server 路由调用
 * 设计思路：置取消标记（runner 消费）→ 结算非终态行为 skipped（手动停止）→ 关窗中止运行中会话 → 清开窗登记
 */
import { todayStr, localWallNow, type AppDb } from '../infrastructure/db'
import type { Logger } from '../infrastructure/logger'
import type { CancelRegistry } from './cancel-registry'

export interface TaskStopperDeps {
  db: AppDb
  registry: CancelRegistry
  logger: Logger
  /** 关闭比特浏览器窗口（中止正在跑的会话；失败忽略） */
  closeBrowser(bitbrowserId: string): Promise<void>
  /** 取开窗登记（停止后清理用） */
  getOpenWindow(bitbrowserId: string): Promise<{ http: string } | null>
  /** 清开窗登记 */
  clearOpenWindow(bitbrowserId: string): Promise<void>
}

export interface TaskStopper {
  /** 停止某窗口某任务当天的在途运行；返回是否确实处理了在途（false = 无在途/窗口不存在） */
  stop(profileId: number, taskKey: string): Promise<boolean>
}

const TERMINAL = ['success', 'failed', 'captcha_failed', 'skipped']

export function createTaskStopper(deps: TaskStopperDeps): TaskStopper {
  const { db, registry, logger } = deps
  return {
    async stop(profileId: number, taskKey: string): Promise<boolean> {
      const date = todayStr()
      const profile = (await db.listProfiles(false)).find((p) => p.id === profileId)
      if (!profile) return false
      const existing = await db.getLatestRun(profileId, taskKey, date)
      const terminal = existing ? TERMINAL.includes(existing.status) : true
      if (!existing || terminal) return false
      registry.request(profileId, taskKey)
      await db.upsertRun(profileId, taskKey, date, existing.slot, 'skipped', { error: '手动停止', finishedAt: localWallNow() })
      await deps.closeBrowser(profile.bitbrowserId).catch(() => {})
      const row = await deps.getOpenWindow(profile.bitbrowserId).catch(() => null)
      if (row) await deps.clearOpenWindow(profile.bitbrowserId).catch(() => {})
      logger.info({ profile: profile.name, task: taskKey }, '已请求停止任务')
      return true
    },
  }
}
```

```ts
/**
 * @swagger
 * /api/batches/{id}/stop:
 *   post:
 *     summary: 停止批次内全部在途运行（落 skipped「手动停止」）
 *     parameters:
 *       - in: path
 *         name: id
 *         required: true
 *         schema: { type: integer }
 *     responses:
 *       '200':
 *         description: 已停止数量
 *         content:
 *           application/json:
 *             schema:
 *               type: object
 *               properties:
 *                 code: { type: integer, example: 0 }
 *                 message: { type: string, example: ok }
 *                 data:
 *                   type: object
 *                   properties:
 *                     stopped: { type: integer }
 *       '404':
 *         description: 批次不存在（业务码 40405）
 *
 * /api/runs/{id}/stop:
 *   post:
 *     summary: 停止单个运行行（落 skipped「手动停止」）
 *     parameters:
 *       - in: path
 *         name: id
 *         required: true
 *         schema: { type: integer }
 *     responses:
 *       '200':
 *         description: 是否停止
 *         content:
 *           application/json:
 *             schema:
 *               type: object
 *               properties:
 *                 code: { type: integer, example: 0 }
 *                 message: { type: string, example: ok }
 *                 data:
 *                   type: object
 *                   properties:
 *                     stopped: { type: boolean }
 *       '404':
 *         description: 运行记录不存在（业务码 40409）
 */
  router.post('/batches/:id/stop', asyncHandler(async (req, res) => {
    const id = Number(req.params.id)
    if (!Number.isInteger(id) || id <= 0) throw new HttpError(400, ERROR_CODES.INVALID_ARGUMENT, '批次 id 必须为正整数')
    const batch = await deps.db.getBatch(id)
    if (!batch) throw new HttpError(404, ERROR_CODES.BATCH_NOT_FOUND, '批次不存在')
    const runs = await deps.db.listRunsForBatch(id)
    let stopped = 0
    for (const r of runs) {
      if (['success', 'failed', 'captcha_failed', 'skipped'].includes(r.status)) continue
      if (await deps.stopper.stop(r.profileId, r.taskKey)) stopped++
    }
    ok(res, { stopped })
  }))
  router.post('/runs/:id/stop', asyncHandler(async (req, res) => {
    const id = Number(req.params.id)
    if (!Number.isInteger(id) || id <= 0) throw new HttpError(400, ERROR_CODES.INVALID_ARGUMENT, '运行 id 必须为正整数')
    const run = await deps.db.getRunById(id)
    if (!run) throw new HttpError(404, ERROR_CODES.RUN_NOT_FOUND, '运行记录不存在')
    ok(res, { stopped: await deps.stopper.stop(run.profileId, run.taskKey) })
  }))
```

```ts
export function useStopBatch() {
  const { message } = App.useApp()
  const queryClient = useQueryClient()
  return useMutation({
    mutationFn: (id: number) => stopBatch(id),
    onSuccess: (res) => {
      message.success(res.stopped > 0 ? `已停止 ${res.stopped} 个运行` : '无在途运行')
      queryClient.invalidateQueries({ queryKey: ['batches'] })
      queryClient.invalidateQueries({ queryKey: ['batchDetail'] })
    },
    onError: (e) => message.error(errMsg(e)),
  })
}

export function useStopRun() {
  const { message } = App.useApp()

      queryClient.invalidateQueries({ queryKey: ['batchDetail'] })
      queryClient.invalidateQueries({ queryKey: ['batches'] })
    },
```
